"""The generated Streamlit app of a Snowflake Native App: one tab per screen of config/screens.json.
A chat screen answers with the agent loop on Cortex and records each answer's usage in
APP_DATA.USAGE before showing it; a table screen runs its tool with the form's inputs and shows
the rows. Chat text is shown as plain text, never as Markdown. Errors show only a stop code or the generic upstream error, never exception text."""

import datetime
import json
from pathlib import Path

import streamlit as st
from snowflake.snowpark.context import get_active_session

import agent_loop
import backend_snowflake

HERE = Path(__file__).resolve().parent
CONFIG = HERE / "config"
MAX_MESSAGE = 4000
UPSTREAM = "upstream_failed: the model or the warehouse did not answer"

session = get_active_session()
config = agent_loop.load_config(CONFIG)
screens = json.loads((CONFIG / "screens.json").read_text(encoding="utf-8"))
agent = config["agent"]
model = agent["cortex_model"]
call_model = backend_snowflake.call_model(session, model)
run_sql = backend_snowflake.run_sql(session)


def chat(index, screen):
    history = st.session_state.setdefault(f"chat-{index}", [])
    for role, text in history:
        st.chat_message(role).text(text)
    question = st.chat_input(screen.get("placeholder") or "", key=f"ask-{index}", max_chars=MAX_MESSAGE)
    if not question:
        return
    st.chat_message("user").text(question)
    try:
        answer = agent_loop.ask(call_model, run_sql, config, question)
        backend_snowflake.record_usage(
            session, agent["release_digest"], model, answer["prompt_tokens"], answer["completion_tokens"]
        )
    except agent_loop.Stop as stop:
        st.error(f"The run stopped: {stop.code}")
        return
    except agent_loop.UpstreamFailed:
        st.error(UPSTREAM)
        return
    st.chat_message("assistant").text(answer["answer"])
    history += [("user", question), ("assistant", answer["answer"])]


def field(index, name, prop):
    key = f"input-{index}-{name}"
    kind = prop.get("type")
    if kind == "integer":
        return st.number_input(name, value=None, step=1, key=key)
    if kind == "number":
        return st.number_input(name, value=None, key=key)
    if kind == "boolean":
        return st.checkbox(name, key=key)
    if prop.get("format") == "date":
        value = st.date_input(name, value=None, key=key)
        return value.isoformat() if isinstance(value, datetime.date) else None
    return st.text_input(name, key=key) or None


def table(index, screen):
    tool = config["tools"][screen["tool"]]
    props = tool["input_schema"].get("properties") or {}
    with st.form(f"form-{index}"):
        values = {name: field(index, name, props.get(name) or {}) for name in screen["inputs"]}
        submitted = st.form_submit_button("Run")
    if not submitted:
        return
    arguments = {name: value for name, value in values.items() if value is not None}
    try:
        result = agent_loop.run_tool(run_sql, tool, arguments)
    except agent_loop.Stop as stop:
        st.error(f"The run stopped: {stop.code}")
        return
    except agent_loop.UpstreamFailed:
        st.error(UPSTREAM)
        return
    names = [c["name"] for c in result["columns"]]
    st.dataframe([dict(zip(names, row)) for row in result["rows"]], hide_index=True)
    if result.get("truncated"):
        st.caption(f"truncated: showing the first {result['row_count']} rows")


for index, (screen, tab) in enumerate(zip(screens, st.tabs([s["title"] for s in screens]))):
    with tab:
        if screen["kind"] == "chat":
            chat(index, screen)
        else:
            table(index, screen)
